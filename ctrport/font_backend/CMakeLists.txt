cmake_minimum_required(VERSION 3.20)
project(CTRFontBackend LANGUAGES C CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
# Existing pinned vendored FreeType; choose GPL-2.0-or-later, not FTL.
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
foreach(dependency ZLIB BZIP2 PNG HARFBUZZ BROTLI)
  set(FT_DISABLE_${dependency} ON CACHE BOOL "" FORCE)
endforeach()
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../../engine/shadPS4/externals/freetype" "${CMAKE_CURRENT_BINARY_DIR}/freetype")
add_executable(font_backend_control control.cpp)
target_link_libraries(font_backend_control PRIVATE freetype)
target_compile_features(font_backend_control PRIVATE cxx_std_17)
if(MSVC)
 target_compile_options(font_backend_control PRIVATE /W4 /WX)
endif()

add_library(ctr_font_backend STATIC backend.cpp)
target_link_libraries(ctr_font_backend PUBLIC freetype)
target_compile_features(ctr_font_backend PRIVATE cxx_std_17)
if(MSVC)
 target_compile_options(ctr_font_backend PRIVATE /W4 /WX)
endif()
target_link_libraries(font_backend_control PRIVATE ctr_font_backend)
