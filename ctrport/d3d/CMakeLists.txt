# ctr_d3d: engine-state D3D11 backend (no shadPS4 sources), and ctr_hle (ctrport/hle) so that the
# backend runs without the ctr_gpu build tree. Build with ctrport/d3d/build.ps1 (clang-cl: the
# guest entry points use the SysV calling convention).
cmake_minimum_required(VERSION 3.24)
project(ctr_d3d LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if (NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE RelWithDebInfo)
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreadedDLL")
if (NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    message(FATAL_ERROR "ctr_d3d needs clang-cl (sysv_abi guest entry points)")
endif()
add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS _WIN32_WINNT=0x0A00 WINVER=0x0A00)

get_filename_component(LLVM_BIN ${CMAKE_CXX_COMPILER} DIRECTORY)
file(GLOB CLANG_RT ${LLVM_BIN}/../lib/clang/*/lib/windows/clang_rt.builtins-x86_64.lib)

file(GLOB D3D_SOURCES CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/*.cpp)
add_library(ctr_d3d SHARED ${D3D_SOURCES})
target_compile_options(ctr_d3d PRIVATE /EHsc -Wno-unused-function -Wno-unused-variable -Wno-unused-command-line-argument)
# SDL3 (zlib licence) for PlayStation controllers, from the same pinned externals the GPU library uses
set(SDL_SHARED OFF CACHE BOOL "" FORCE)
set(SDL_STATIC ON CACHE BOOL "" FORCE)
set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
set(SDL_AVX512F OFF CACHE BOOL "" FORCE)
add_subdirectory(${CMAKE_CURRENT_SOURCE_DIR}/../../engine/shadPS4/externals/sdl3 sdl3 EXCLUDE_FROM_ALL)
target_link_libraries(ctr_d3d PRIVATE user32 gdi32 d3d11 dxgi d3dcompiler SDL3::SDL3-static ${CLANG_RT})
# FSR 1 shader sources (fsr/: AMD FidelityFX headers, MIT), compiled by the renderer at startup
set(FSR_INC ${CMAKE_CURRENT_BINARY_DIR}/fsr_sources.inc)
file(WRITE ${FSR_INC} "// generated by ctrport/d3d/CMakeLists.txt from ctrport/d3d/fsr\n")
foreach(f fsr1.hlsl ffx_a.h ffx_fsr1.h)
    file(READ ${CMAKE_CURRENT_SOURCE_DIR}/fsr/${f} FSR_TEXT)
    file(APPEND ${FSR_INC} "{\"${f}\", R\"ffx(${FSR_TEXT})ffx\"},\n")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/fsr/${f})
endforeach()
target_include_directories(ctr_d3d PRIVATE ${CMAKE_CURRENT_BINARY_DIR})

file(GLOB HLE_SOURCES CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/../hle/*.cpp)
add_library(ctr_hle SHARED ${HLE_SOURCES})
target_compile_definitions(ctr_hle PRIVATE CTR_HLE_BUILD)
target_compile_options(ctr_hle PRIVATE -Wno-unused-command-line-argument)
target_link_libraries(ctr_hle PRIVATE ${CLANG_RT})

# ctr_shaders.exe: the shader translator (GCN -> HLSL -> DXBC) run on the user's own dump
add_executable(ctr_shaders shaders/ctr_shaders.cpp shaders/variants.cpp)
target_compile_options(ctr_shaders PRIVATE /EHsc -Wno-unused-function -Wno-unused-variable -Wno-unused-command-line-argument)
target_link_libraries(ctr_shaders PRIVATE bcrypt)
# the settings headers it shares with the launcher are C++17 (u8string returns std::string)
set_target_properties(ctr_shaders PROPERTIES CXX_STANDARD 17)
# the GPU library's shader-override contracts it reuses are C++20
set_source_files_properties(shaders/variants.cpp PROPERTIES COMPILE_OPTIONS "/std:c++20")
